59void EdgeSwapOptions::set_mode(EdgeSwapMode mode)
60{
61 std::string r;
62 switch (mode) {
63 case EdgeSwapMode::AMIPS: r = "amips"; break;
64 case EdgeSwapMode::Valence: r = "valence"; break;
65 default: break;
66 }
67
68 parameters["mode"] = r;

70EdgeSwapMode EdgeSwapOptions::mode() const
71{
72 if (!parameters.contains("mode")) {
73 wmtk::logger().debug("Edge swap did not have a mode, defaulting to valence");
74 return EdgeSwapMode::Valence;
75 }
76 const std::string swap_name = parameters["mode"].get<std::string>();
77 if (swap_name == "amips") {
78 return EdgeSwapMode::AMIPS;
79 } else if (swap_name == "valence") {
80 return EdgeSwapMode::Valence;
81 } else {
82 throw std::runtime_error(
83 fmt::format(
84 "Expected edge_swap_mode to be one of [amips,valence], got [{}]",
85 swap_name));
86 }
