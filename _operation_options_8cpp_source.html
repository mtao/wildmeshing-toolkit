68void EdgeSwapOptions::set_mode(EdgeSwapMode mode)
69{
70 std::string r;
71 switch (mode) {
72 case EdgeSwapMode::AMIPS: r = "amips"; break;
73 case EdgeSwapMode::Valence: r = "valence"; break;
74 default: break;
75 }
76
77 parameters["mode"] = r;
78}
79EdgeSwapMode EdgeSwapOptions::mode() const
80{
81 if (!parameters.contains("mode")) {
82 wmtk::logger().debug("Edge swap did not have a mode, defaulting to valence");
83 return EdgeSwapMode::Valence;
84 }
85 const std::string swap_name = parameters["mode"].get<std::string>();
86 if (swap_name == "amips") {
87 return EdgeSwapMode::AMIPS;
88 } else if (swap_name == "valence") {
89 return EdgeSwapMode::Valence;
90 } else {
91 throw std::runtime_error(
92 fmt::format(
93 "Expected edge_swap_mode to be one of [amips,valence], got [{}]",
94 swap_name));
95 }
96}
