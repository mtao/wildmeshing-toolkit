38// compute the length relative to the bounding box diagonal
39double relative_to_absolute_length(
40 const attribute::MeshAttributeHandle& position,
41 const double length_rel)
42{
43 if (!position.is_valid()) {
44 throw std::runtime_error(
45 "Could not convert relative to absolute length because position attr was invalid");
46 }
47 auto pos = position.mesh().create_const_accessor<double>(position);
48 const auto vertices = position.mesh().get_all(PrimitiveType::Vertex);
49 Eigen::AlignedBox<double, Eigen::Dynamic> bbox(pos.dimension());
50
51
52 for (const auto& v : vertices) {
53 bbox.extend(pos.const_vector_attribute(v));
54 }
55
56 const double diag_length = bbox.sizes().norm();
57 wmtk::logger().debug(
58 "computed absolute target length using relative factor {} on bbox diagonal {}",
59 length_rel,
60 diag_length);
61
62 return length_rel * diag_length;
63}

66double IsotropicRemeshingOptions::get_absolute_length(const multimesh::MeshCollection& mc) const
67{
68 double length = length_abs;
69 if (length_abs <= 0) {
70 if (length_rel <= 0) {
71 throw std::runtime_error("Either absolute or relative length must be set!");
72 }
73 auto position_attr = multimesh::utils::get_attribute(mc, position_attribute);
74 length = relative_to_absolute_length(position_attr, length_rel);
75 } else {
76 wmtk::logger().debug(
77 "get_absolute_length using absolute length value {} {}",
78 length,
79 length_abs);
80 }
81 return length;
