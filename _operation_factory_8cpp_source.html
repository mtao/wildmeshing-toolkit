20std::shared_ptr<wmtk::operations::Operation> default_add_operation(
21 Configurator& c,
22 const nlohmann::json& js)
23{
24 auto opts = js.template get<S>();
25 auto& m = c.get_mesh<MeshType>(opts.mesh_path);
26 auto r = std::make_shared<T>(m);
27
28 if (opts.priority) {
29 opts.priority.assign_to(c.meshes(), *r);
30 }
31 for (const auto& [name, inv] : opts.invariants) {
32 r->add_invariant(c.create_invariant(name, inv));
33 }
34 return r;

36std::shared_ptr<wmtk::operations::Operation> add_swap_operation(
37 Configurator& c,
38 const nlohmann::json& js)
39{
40 auto opts = js.template get<EdgeSwapOptions>();
41 auto& m_ = c.get_mesh<wmtk::Mesh>(opts.mesh_path);
42
43 int8_t dim = m_.top_cell_dimension();
44 std::shared_ptr<wmtk::operations::composite::EdgeSwap> r;
45 assert(dim == 2 || dim == 3);
46 if (dim == 2) {
47 auto& m = c.get_mesh<wmtk::TriMesh>(opts.mesh_path);
48 r = std::make_shared<wmtk::operations::composite::TriEdgeSwap>(m);
49 } else {
50 auto& m = c.get_mesh<wmtk::TetMesh>(opts.mesh_path);
51 r = std::make_shared<wmtk::operations::composite::TetEdgeSwap>(m);
52 }
53
54
55 if (opts.priority) {
56 opts.priority.assign_to(c.meshes(), *r);
57 }
58 for (const auto& [name, inv] : opts.invariants) {
59 r->add_invariant(c.create_invariant(name, inv));
60 }
61 return r;
62}
